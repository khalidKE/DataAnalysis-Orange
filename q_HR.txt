🧮 Important Measures (Power BI DAX)
Headcount
Total Employees = COUNT('HR'[Employee ID])

Active Employees
Active = CALCULATE(COUNT('HR'[Employee ID]), 'HR'[EmployeeStatus] = "Active")

Turnover Rate
Turnover % = 
DIVIDE(
    COUNTROWS(FILTER('HR', NOT(ISBLANK('HR'[ExitDate])))),
    [Total Employees]
)

Avg Satisfaction
Avg Satisfaction = AVERAGE('HR'[Satisfaction Score])

Avg Performance
Avg Performance = AVERAGE('HR'[Current Employee Rating])

Training Cost
Total Training Cost = SUM('HR'[Training Cost])
